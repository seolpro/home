/** ========= COMMON UTIL ========= */
function getSS_() {

                         : SpreadsheetApp.getActiveSpreadsheet();
}
function getSheet_(name) {
  const sh = getSS_().getSheetByName(name);
  if (!sh) throw new Error(`시트를 찾을 수 없습니다: ${name}`);
  return sh;
}
function v_(x){ return x == null ? '' : String(x).trim(); }
function _idx_(header){ return header.reduce((m,h,i)=>(m[String(h).trim()]=i,m),{}); }
function escapeHtml_(s) {
  return String(s).replace(/[&<>"']/g, m=>({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[m]));
}

/** ========= SCHEMA-SHEET HEADER ENSURE =========
 * schema 시트의 헤더(formKey|order|label|type|name|options|required) 보장
 */
function ensureSchemaHeader_() {
  const sh = getSheet_(CONFIG.SCHEMA_SHEET);
  const header = ['formKey','order','label','type','name','options','required'];
  const range = sh.getRange(1, 1, 1, header.length);
  const cur = range.getValues()[0].map(String);
  const needWrite = header.some((h,i) => cur[i] !== h);
  if (needWrite) {
    range.setValues([header]);
    sh.setFrozenRows(1);
  }
}

/** ========= LOAD SCHEMA (for form runtime) =========
 * meta(title/submitText/loadingText) 포함 로딩
 */
function loadSchema_(formKey) {
  ensureSchemaHeader_();
  const sh = getSheet_(CONFIG.SCHEMA_SHEET);
  const values = sh.getDataRange().getValues();
  const header = values.shift();
  const idx = _idx_(header);
  ['formKey','order','label','type','name','options','required']
    .forEach(h=>{ if (idx[h]==null) throw new Error(`schema 헤더 누락: ${h}`); });

  const all = values
    .filter(r => v_(r[idx.formKey]) === formKey)
    .map(r => ({
      order   : Number(r[idx.order] || 0),
      label   : v_(r[idx.label]),
      type    : v_(r[idx.type]).toLowerCase(),
      name    : v_(r[idx.name]) || v_(r[idx.label]),
      options : v_(r[idx.options]),
      required: /^y/i.test(v_(r[idx.required])),
    }))
    .sort((a,b)=>a.order-b.order);

  const meta = {};
  all.filter(r => r.type==='meta' && ['title','submitText','loadingText'].includes(r.name))
     .forEach(r => { meta[r.name] = r.label; });

  // 폼 렌더링용 스키마: meta 제외
  const fields = all.filter(r => r.type !== 'meta');
  return { fields, meta };
}

/** ========= DATA-SHEET HEADER ENSURE (보강판) =========
 * - 비어있으면: [timestamp] + schema.name[]
 * - 기존 헤더가 있으면: schema/payload에 있는 누락 컬럼을 오른쪽에 추가
 * - timestamp 없으면 맨 앞(기본) 또는 맨 뒤에 추가 (옵션)
 */
function ensureHeader_(dataSh, schema, payload, opts) {
  opts = opts || {};
  const ADD_PREFIX_AT_END = !!opts.putTimestampAtEnd;
  const MUST_PREFIX = ADD_PREFIX_AT_END ? [] : ['timestamp'];

  const schemaNames = (schema || []).map(f => String(f && f.name || '').trim()).filter(Boolean);
  const payloadKeys = Object.keys(payload || {});
  const needed = dedupe([].concat(MUST_PREFIX, schemaNames, payloadKeys));

  const lastRow = dataSh.getLastRow();
  if (lastRow < 1) {
    const header = ADD_PREFIX_AT_END ? dedupe([].concat(schemaNames, payloadKeys, ['timestamp']))
                                     : dedupe([].concat(['timestamp'], schemaNames, payloadKeys));
    dataSh.getRange(1, 1, 1, header.length).setValues([header]);
    try { dataSh.setFrozenRows(1); } catch(_) {}
    return header;
  }

  let header = dataSh.getRange(1, 1, 1, dataSh.getLastColumn()).getValues()[0].map(String);
  if (header.every(h => !String(h || '').trim())) {
    const newHeader = ADD_PREFIX_AT_END ? dedupe([].concat(schemaNames, payloadKeys, ['timestamp']))
                                        : dedupe([].concat(['timestamp'], schemaNames, payloadKeys));
    dataSh.getRange(1, 1, 1, newHeader.length).setValues([newHeader]);
    try { dataSh.setFrozenRows(1); } catch(_) {}
    return newHeader;
  }

  const hasTimestamp = header.some(h => String(h).trim() === 'timestamp');
  if (!hasTimestamp) {
    if (ADD_PREFIX_AT_END) {
      header.push('timestamp');
      dataSh.getRange(1, 1, 1, header.length).setValues([header]);
    } else {
      header = ['timestamp'].concat(header);
      dataSh.insertColumns(1, 1);
      dataSh.getRange(1, 1, 1, header.length).setValues([header]);
    }
  }

  const missing = needed.filter(col => !header.includes(col));
  if (missing.length) {
    const startCol = header.length + 1;
    dataSh.getRange(1, startCol, 1, missing.length).setValues([missing]);
    header = header.concat(missing);
  }

  try { dataSh.setFrozenRows(1); } catch(_) {}
  return header;
}
function dedupe(arr) {
  const seen = Object.create(null), out = [];
  for (const v of arr) { const k = String(v); if (!k) continue; if (!seen[k]) { seen[k]=1; out.push(k); } }
  return out;
}

/**
 * data:image/*;base64,.... 형태의 dataURL -> Blob 변환
 */
function dataUrlToBlob_(dataUrl, fallbackMime) {
  if (typeof dataUrl !== 'string') throw new Error('dataUrl must be a string');
  const m = dataUrl.match(/^data:([^;]+);base64,(.+)$/);
  if (!m) throw new Error('Invalid data URL. Expect "data:<mime>;base64,<data>"');
  const mime = m[1] || fallbackMime || 'image/png';
  const bytes = Utilities.base64Decode(m[2]);
  return Utilities.newBlob(bytes, mime);
}

/**
 * 파일명에서 드라이브 금지/문제 문자 제거
 */
function sanitizeFileName_(name) {
  const safe = String(name || '')
    .replace(/[\x00-\x1f"*/:<>?\\|]+/g, '-') // 금지문자
    .replace(/\s+/g, ' ')
    .trim();
  return safe || 'signature';
}

/**
 * 서명 이미지를 드라이브에 저장하고 미리보기 가능한 URL 반환
 * @param {string} dataUrl - data:image/png;base64,... 형식
 * @param {string=} filename - 확장자 없이 넣어도 자동으로 .png 부여
 * @param {Object=} opts
 * @param {string=} opts.folderId - 저장 폴더
 * @param {boolean=} opts.makePublic - true면 링크공개(Anyone with link)
 * @param {string=} opts.timezone - 기본: Script TZ
 */
function saveSignatureToDrive_(dataUrl, filename, opts) {
  opts = opts || {};
  const tz = opts.timezone || Session.getScriptTimeZone() || 'Asia/Seoul';

  // 1) Blob 생성 (mime 지정)
  const blob = dataUrlToBlob_(dataUrl, 'image/png');

  // 2) 파일명 구성(+ 확장자)
  const ts = Utilities.formatDate(new Date(), tz, 'yyyyMMdd_HHmmss');
  const base = sanitizeFileName_(filename) || ('signature_' + ts);
  const ext = blob.getContentType() && blob.getContentType().indexOf('png') >= 0 ? '.png' : '';
  blob.setName(base + ext);

  // 3) 파일 생성
  const file = opts.folderId
    ? DriveApp.getFolderById(opts.folderId).createFile(blob)
    : DriveApp.createFile(blob);

  // 4) (선택) 링크공개 -> <img src="..."> 바로 사용 가능
  if (opts.makePublic) {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  }

  // 5) 리턴 값 구성
  const id = file.getId();
  // 미리보기/임베드 용도(권한 필요 시 표시 안 됨)
  const viewUrl = 'https://drive.google.com/uc?export=view&id=' + id;
  // UI(드라이브 웹) 링크
  const uiUrl = file.getUrl();

  return {
    id,
    name: file.getName(),
    mimeType: file.getMimeType(),
    createdTime: Utilities.formatDate(new Date(file.getDateCreated()), tz, 'yyyy-MM-dd HH:mm:ss'),
    // <img src="...">에 바로 쓰려면 makePublic=true 이어야 합니다.
    url: viewUrl,          // 임베드/미리보기 권장 URL
    uiUrl: uiUrl           // 드라이브 UI 페이지
  };
}

/** ========= SAVE ========= */
function saveData_(formKey, payload, idempotencyKey) {
  const dataSh = getSheet_(CONFIG.DATA_SHEET);
  const { fields: schema } = loadSchema_(formKey);

  // === 타입 Map ===
  const typeMap = {};
  schema.forEach(f => typeMap[f.name] = (f.type || '').toLowerCase());

  // === 필수 검증 (file은 id/url/base64 중 1개 이상 필수) ===
  schema.filter(f => f.required).forEach(f => {
    const key = f.name || f.label;
    const t = (f.type || '').toLowerCase();
    const val = payload[key];

    if (t === 'file') {
      const ok = val && typeof val === 'object' && (val.id || val.url || val.base64);
      if (!ok) throw new Error(`필수 입력 누락(파일): ${f.label}`);
    } else if (t === 'checkbox') {
      if (String(val || '') !== 'Y') throw new Error(`필수 입력 누락: ${f.label}`);
    } else if (t === 'checkboxes') {
      if (!String(val || '').trim()) throw new Error(`필수 입력 누락: ${f.label}`);
    } else {
      if (!v_(val)) throw new Error(`필수 입력 누락: ${f.label}`);
    }
  });

  // === 서명 처리 (기존 그대로) ===
  Object.keys(payload).forEach(k => {
    if (typeMap[k] === 'signature' && /^data:image\/(png|jpeg);base64,/.test(String(payload[k] || ''))) {
      if (CONFIG.SIGNATURE.saveToDrive) {
        const saved = saveSignatureToDrive_(payload[k], 'sig_' + k + '.png');
        payload[k] = (CONFIG.SIGNATURE.store === 'id')  ? saved.id
                   : (CONFIG.SIGNATURE.store === 'url') ? saved.url
                   : payload[k];
      } else {
        if (String(payload[k]).length > 45000) throw new Error('서명 이미지가 너무 큽니다. 드라이브 저장 옵션을 사용하세요.');
      }
    }
  });

  // === 파일 처리: base64가 남아있다면 드라이브에 저장하고, 저장형식(url/id/json)에 맞게 치환 ===
  Object.keys(payload).forEach(k => {
    if (typeMap[k] !== 'file') return;
    payload[k] = normalizeFileField_(formKey, k, payload[k]); // 문자열 or JSON으로 치환
  });

  // === 헤더 보장 후 append ===
  const header = ensureHeader_(dataSh, schema, payload);
  const now = new Date();
  const rowValues = header.map((h, i) => {
    const key = String(h || '').trim();
    if (!key) return '';
    if (i === 0 && key.toLowerCase() === 'timestamp') return now;

    const t = typeMap[key];
    const val = payload[key];

    if (t === 'file') {
      // 이미 normalizeFileField_에서 문자열/JSON으로 변환됨
      return val || '';
    }
    return v_(val);
  });

  dataSh.appendRow(rowValues);
  return { rowNumber: dataSh.getLastRow() };
}

/** 파일 셀값을 설정값에 맞게 문자열/JSON으로 정규화
 *  - {base64,name,type,size}만 온 경우: Drive 저장 후 {id,url,...} 확보
 *  - CONFIG.FILE.store: 'url' | 'id' | 'json' (기본 'url')
 */
function normalizeFileField_(formKey, fieldName, fileVal) {
  if (!fileVal) return '';

  // 이미 문자열인 경우(하위 호환)
  if (typeof fileVal === 'string') return fileVal;

  // 객체가 아니면 문자열화
  if (typeof fileVal !== 'object') return String(fileVal);

  // base64가 있으면 지금 저장
  if (fileVal.base64) {
    try {
      const saved = saveBase64FileToDrive_(formKey, fileVal, fieldName);
      fileVal = Object.assign({}, fileVal, saved); // {id,url,name,type,size} 병합
      // base64는 무거우므로 제거
      delete fileVal.base64;
    } catch (e) {
      return JSON.stringify({ error: 'file_save_failed', message: String(e) });
    }
  }

  // 저장 형태 선택
  const mode = (CONFIG.FILE && CONFIG.FILE.store) || 'url';
  if (mode === 'id')   return fileVal.id || '';
  if (mode === 'json') return JSON.stringify(fileVal);
  // 기본: url
  return fileVal.url || fileVal.id || fileVal.name || '';
}

/** Base64 파일을 Drive에 저장하고 {id,url,name,type,size} 반환 */
function saveBase64FileToDrive_(formKey, fileObj, fieldName) {
  const name = sanitizeFilename_(
    fileObj.name ||
    (fieldName + '_' + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Seoul', 'yyyyMMdd_HHmmss') + '.bin')
  );
  const mime = fileObj.type || MimeType.BINARY;
  const bytes = Utilities.base64Decode(String(fileObj.base64));
  const blob = Utilities.newBlob(bytes, mime, name);

  const folder = getOrCreateUploadFolder_(formKey);
  const file = folder.createFile(blob);

  // 필요 시 공개 설정
  // file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

  return {
    id: file.getId(),
    url: 'https://drive.google.com/file/d/' + file.getId() + '/view',
    name: file.getName(),
    type: mime,
    size: fileObj.size || file.getSize()
  };
}

/** 업로드 폴더 획득/생성: 루트/FormsUpload/<formKey> */
function getOrCreateUploadFolder_(formKey) {
  const rootName = 'FormsUpload';
  const parents = DriveApp.getFoldersByName(rootName);
  const root = parents.hasNext() ? parents.next() : DriveApp.createFolder(rootName);

  const kids = root.getFoldersByName(formKey);
  const folder = kids.hasNext() ? kids.next() : root.createFolder(formKey);
  return folder;
}

/** 파일명 안전화 */
function sanitizeFilename_(name) {
  name = String(name || 'upload.bin');
  return name.replace(/[\\\/:*?"<>|#\r\n\t]/g, '_').slice(0, 120);
}


/** ========= RESPOND HELPERS ========= */
function respondJson_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
                       .setMimeType(ContentService.MimeType.JSON);
}

/** ========= META LOAD (title/submitText/loadingText) ========= */
function loadMeta_(formKey) {
  const sh = getSheet_(CONFIG.SCHEMA_SHEET);
  const values = sh.getDataRange().getValues();
  const header = values.shift();
  const idx = _idx_(header);
  const meta = {};
  values.forEach(r => {
    if (v_(r[idx.formKey]) !== formKey) return;
    const type = v_(r[idx.type]).toLowerCase();
    if (type === 'meta') {
      const key = v_(r[idx.name]);
      const val = v_(r[idx.label]) || v_(r[idx.options]);
      if (key) meta[key] = val;
    }
  });
  return meta;
}

/** ========= WEB APP: GET =========
 * ?action=schema&formKey=...
 */

function doGet(e) {
  try {
    const action = (e.parameter.action || '').toLowerCase();

    // 1) JSON API: 스키마 조회
    if (action === 'schema') {
      const formKey = e.parameter.formKey || '';
      if (!formKey) return respondJson_({ ok:false, error:'formKey required' });
      const { fields, meta } = loadSchema_(formKey);
      return respondJson_({ ok:true, schema: fields, meta });
    }

    // 2) HTML 페이지: SchemaBuild.html (iframe 임베드용)
    if (action === 'builder') {
      const t = HtmlService.createTemplateFromFile('SchemaBuild');
      // 필요 시 쿼리로 전달: ?action=builder&formKey=join2025
      t.formKey = e.parameter.formKey || '';

      return t.evaluate()
        .setTitle('Schema Builder')
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) // 외부 도메인 iframe 허용
        .addMetaTag('viewport', 'width=device-width, initial-scale=1');
    }

    // (선택) 기본값을 HTML로 돌리고 싶다면 아래 한 줄로 교체해도 됩니다.
    // return HtmlService.createHtmlOutput('no action').setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    return respondJson_({ ok:false, error: 'no action' });
  } catch (err) {
    return respondJson_({ ok:false, error:String(err) });
  }
}

/** ========= WEB APP: POST =========
 * JSON 또는 x-www-form-urlencoded(payload=JSON) 모두 지원
 */
function doPost(e) {
  try {
    // 바디 파싱
    const ctype = (e.postData && e.postData.type) ? e.postData.type : '';
    let body = {};
    if (/application\/json/i.test(ctype)) {
      body = JSON.parse(e.postData.contents || '{}');
    } else if (/application\/x-www-form-urlencoded/i.test(ctype)) {
      body = JSON.parse(e.parameter.payload || '{}');
    } else {
      const raw = (e.postData && e.postData.contents) || e.parameter.payload || '{}';
      try { body = JSON.parse(raw); } catch (_) { body = {}; }
    }

    const formKey = v_(body.formKey);
    const payload = body.payload || {};
    const idk = v_(body.idempotencyKey || '');
    if (!formKey) throw new Error('formKey is required');

    // 저장
    const saved = saveData_(formKey, payload, idk);
    const savedRow = (saved && (saved.row ?? saved.rowNumber ?? saved.savedRow ?? saved.lastRow)) || null;

    // 알림(이메일+알림톡)
    try { notifyOnSubmit_(formKey, payload, savedRow); }
    catch (nerr) { console.error('[notifyOnSubmit_] 실패:', nerr); }

    return respondJson_({ ok: true, saved, row: savedRow });
  } catch (err) {
    return respondJson_({ ok: false, error: String(err) });
  }
}

function saveNotifyConfig(cfg) {
  if (!cfg || typeof cfg !== 'object') return { ok:false, error:'invalid cfg' };
  const sp = PropertiesService.getScriptProperties();

  if ('ENABLE_EMAIL_NOTIFY' in cfg)    sp.setProperty('ENABLE_EMAIL_NOTIFY', String(!!cfg.ENABLE_EMAIL_NOTIFY));
  if ('ENABLE_ALIMTALK_NOTIFY' in cfg) sp.setProperty('ENABLE_ALIMTALK_NOTIFY', String(!!cfg.ENABLE_ALIMTALK_NOTIFY));

  const setIfProvided = (k, def='') => { if (k in cfg) sp.setProperty(k, String(cfg[k] || def)); };
  setIfProvided('ADMIN_EMAILS');
  setIfProvided('EMAIL_SUBJECT_PREFIX', '');
  setIfProvided('EMAIL_SENDER_NAME', '');
  setIfProvided('SUBMITTER_PHONE_FIELD', '');

  setIfProvided('PPURIO_ACCOUNT');
  setIfProvided('PPURIO_SENDER_KEY');
  setIfProvided('PPURIO_FROM');
  setIfProvided('PPURIO_TEMPLATE_CODE');
  setIfProvided('PPURIO_TEMPLATE_MESSAGE', '안녕하세요 #{name}님, #{*1*} 가(이) 접수되었습니다.');

  if ('PPURIO_PASSWORD' in cfg) {
    const pwd = String(cfg.PPURIO_PASSWORD || '');
    if (pwd) sp.setProperty('PPURIO_PASSWORD', pwd);
  }
  if (cfg && cfg.reset_PPURIO_PASSWORD) sp.deleteProperty('PPURIO_PASSWORD');

  return { ok:true };
}

/** ========= TEMPLATE & PHONE HELPERS ========= */
function renderTemplate_(tmpl, data) {
  return String(tmpl||'').replace(/#\{(\w+)\}/g,(_,k)=> data && data[k]!=null ? String(data[k]) : '');
}

function pickNameFromPayload_(payload) {
  if (!payload) return '';
  const candidates = ['name','성명','이름','신청자','대표자','담당자','신청자명'];
  for (const k of candidates) {
    if (payload[k]) return String(payload[k]).trim();
  }
  const hit = Object.keys(payload).find(k => /name|이름|성명/i.test(k) && String(payload[k]).trim());
  return hit ? String(payload[hit]).trim() : '';
}
function pickPhoneFromPayload_(payload, preferredKey) {
  if (!payload) return '';
  const keys = [preferredKey,'tel','phone','mobile','cell','휴대폰','연락처'].filter(Boolean);
  for (const k of keys) { const v = payload[k]; if (v) return String(v).replace(/[^\d]/g,''); }
  const any = Object.values(payload).map(v=>String(v)).join(' ');
  const m = any.match(/01[016789]\d{3,4}\d{4}/);
  return m ? m[0] : '';
}

// 숫자만 남기기
function digitsOnly_(s) {
  return String(s == null ? '' : s).replace(/[^0-9]/g, '');
}


// 010 형식 변환
function formatPhoneKRMobile_(raw) {
  const orig = String(raw == null ? '' : raw).trim();
  let s = digitsOnly_(orig);

  // 10자리인데 0으로 시작 안 하면 앞에 0 붙이기
  if (s.length === 10 && s[0] !== '0') {
    s = '0' + s;
  }

  // 11자리 01x → 3-4-4 패턴
  if (s.length === 11 && /^01\d/.test(s)) {
    return s.replace(/(\d{3})(\d{4})(\d{4})/, '$1-$2-$3');
  }

  return orig;
}

// 열 이름(A, B, C...)을 번호로 변환
function columnLetterToNumber_(letter) {
  let col = 0;
  letter = letter.toUpperCase();
  for (let i = 0; i < letter.length; i++) {
    col = col * 26 + (letter.charCodeAt(i) - 64);
  }
  return col;
}

// 프롬프트로 열 이름 입력 → 해당 열 전체 변환
function promptAndFormatColumnLetter_() {
  const ui = SpreadsheetApp.getUi();
  const sh = SpreadsheetApp.getActiveSheet();

  const resp = ui.prompt('연락처 변환', '변환할 열 이름을 입력하세요 (예: C, D, AA)', ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;

  const colLetter = resp.getResponseText().trim().toUpperCase();
  if (!/^[A-Z]+$/.test(colLetter)) {
    ui.alert('올바른 열 이름을 입력하세요. 예: C, D, AA');
    return;
  }

  const col = columnLetterToNumber_(colLetter);
  const lastRow = sh.getLastRow();
  if (lastRow < 2) {
    ui.alert('데이터가 없습니다.');
    return;
  }

  const rng = sh.getRange(2, col, lastRow - 1, 1); // 2행부터 마지막 행까지
  const vals = rng.getValues();

  for (let i = 0; i < vals.length; i++) {
    vals[i][0] = formatPhoneKRMobile_(vals[i][0]);
  }

  rng.setValues(vals);
  ui.alert(`열 ${colLetter} 변환 완료`);
}

// schema UI용: 스키마 읽기
function sb_getSchema(formKey) {
  ensureSchemaHeader_(); // ✅ 헤더 자동 보장
  const sh = getSheet_(SCHEMA_SHEET_NAME);
  const values = sh.getDataRange().getValues();
  const header = values.shift();
  const idx = _idx_(header);
  ['formKey','order','label','type','name','options','required'].forEach(h=>{
    if (idx[h]==null) throw new Error(`schema 헤더 누락: ${h}`);
  });

  const keys = Array.from(new Set(values.map(r => String(r[idx.formKey]||'').trim()).filter(Boolean)));

  const allRows = values
    .filter(r => String(r[idx.formKey]||'').trim() === formKey)
    .map(r => ({
      order   : Number(r[idx.order]),
      label   : String(r[idx.label]||'').trim(),
      type    : String(r[idx.type]||'').toLowerCase().trim(),
      name    : String(r[idx.name]||'').trim(),
      options : String(r[idx.options]||'').trim(),
      required: /^y/i.test(String(r[idx.required]||'')),
    }));

  const meta = { title:'', submitText:'', loadingText:'' };
  allRows.filter(r => r.type==='meta' && ['title','submitText','loadingText'].includes(r.name))
         .forEach(r => { meta[r.name] = r.label; });

  const rows = allRows.filter(r => r.type !== 'meta').sort((a,b)=>a.order-b.order);

  return { ok:true, formKeys: keys, rows, meta, allowedTypes: ALLOWED_TYPES, headerFromRow: HEADER_FROM_ROW };
}

// schema UI용: 스키마 저장
function sb_saveSchema(formKey, rows, meta) {
  ensureSchemaHeader_(); // ✅ 헤더 자동 보장
  if (!formKey) throw new Error('formKey가 필요합니다.');
  if (!Array.isArray(rows)) throw new Error('rows 형식 오류');
  meta = meta || {};

  rows.forEach((r,i)=>{
    const rowNo = i+1;
    const type = String(r.type||'').toLowerCase();
    if (isNaN(Number(r.order))) throw new Error(`${rowNo}행: order가 숫자가 아닙니다.`);
    if (!r.label)                throw new Error(`${rowNo}행: label이 비어 있습니다.`);
    if (!ALLOWED_TYPES.includes(type)) throw new Error(`${rowNo}행: type("${type}")이 허용 목록에 없습니다.`);
    if (!r.name)                 throw new Error(`${rowNo}행: name이 비어 있습니다.`);
    if (['select','radio','checkboxes'].includes(type) && !String(r.options||'').trim())
      throw new Error(`${rowNo}행: ${type} 타입은 options가 필요합니다.`);
  });

  const sh = getSheet_(SCHEMA_SHEET_NAME);
  const values = sh.getDataRange().getValues();
  const header = values[0];
  const idx = _idx_(header);

  // 기존 formKey 행 삭제
  for (let r = sh.getLastRow(); r >= 2; r--) {
    const k = String(sh.getRange(r, idx.formKey+1).getValue()||'').trim();
    if (k === formKey) sh.deleteRow(r);
  }

  const toWrite = [];
  const metaTriples = [
    {name:'title',       label: String(meta.title||'')       },
    {name:'submitText',  label: String(meta.submitText||'')  },
    {name:'loadingText', label: String(meta.loadingText||'') },
  ];
  metaTriples.forEach(m => toWrite.push([formKey, 0, m.label, 'meta', m.name, '', 'N']));

  rows.sort((a,b)=>Number(a.order)-Number(b.order));
  rows.forEach(r=> toWrite.push([
      formKey, Number(r.order), String(r.label||''), String(r.type||'').toLowerCase(),
      String(r.name||''), String(r.options||''), r.required ? 'Y' : 'N'
  ]));

  if (toWrite.length) sh.getRange(sh.getLastRow()+1, 1, toWrite.length, 7).setValues(toWrite);
  return { ok:true, count: toWrite.length };
}

/** ====== join 헤더 동기화/서식 (UI 메뉴) ====== */
function syncJoinHeadersFromSchema() {
  const ss = getSS_();
  const schemaSh = ss.getSheetByName(SCHEMA_SHEET_NAME);
  const joinSh   = ss.getSheetByName(JOIN_SHEET_NAME);
  if (!schemaSh) throw new Error(`"${SCHEMA_SHEET_NAME}" 시트를 찾을 수 없습니다.`);
  if (!joinSh)   throw new Error(`"${JOIN_SHEET_NAME}" 시트를 찾을 수 없습니다.`);

  const values = schemaSh.getDataRange().getValues();
  const header = values[0];
  const idx = _idx_(header);
  ['formKey','order','type','name'].forEach(h=>{ if (idx[h]==null) throw new Error(`schema 헤더 누락: ${h}`); });

  const formKey = Browser.inputBox('formKey 입력', '예: join2025', Browser.Buttons.OK_CANCEL);
  if (formKey === 'cancel' || !formKey) return;

  const rows = values.slice(HEADER_FROM_ROW - 1)
    .map(r => ({
      formKey: String(r[idx.formKey]||'').trim(),
      order  : Number(r[idx.order]),
      type   : String(r[idx.type]||'').toLowerCase(),
      name   : String(r[idx.name]||'').trim()
    }))
    .filter(r => r.formKey === formKey && r.type !== 'meta' && !isNaN(r.order) && r.name);

  if (!rows.length) throw new Error(`동기화할 name이 없습니다. (formKey=${formKey}, ${HEADER_FROM_ROW}행 이후)`);

  rows.sort((a,b)=>a.order-b.order);
  const names = rows.map(r=>r.name);

  const needCols = names.length + 1;
  if (joinSh.getMaxColumns() < needCols) joinSh.insertColumnsAfter(joinSh.getMaxColumns(), needCols - joinSh.getMaxColumns());

  if (!joinSh.getRange('A1').getValue()) joinSh.getRange('A1').setValue('timestamp');
  joinSh.getRange(1, 2, 1, names.length).setValues([names]);

  const lastCol = joinSh.getLastColumn();
  const rightStartCol = names.length + 2;
  if (lastCol >= rightStartCol) joinSh.getRange(1, rightStartCol, 1, lastCol - rightStartCol + 1).clearContent();

  ss.toast(`헤더 동기화 완료: ${names.length}개`, '완료', 3);
}

  sh.setFrozenRows(1);
  ss.toast('join 헤더 서식 적용 완료', '완료', 3);
}

/** ========= 이메일 발송 ========= */
function sendEmailNotification_(formKey, payload, savedRow, CFG) {
  // 0) 방어 코드: CFG 및 필수 필드
  if (!CFG) throw new Error('CFG is required');
  const admins = Array.isArray(CFG.ADMIN_EMAILS) ? CFG.ADMIN_EMAILS.filter(Boolean) : [];
  if (admins.length === 0) throw new Error('CFG.ADMIN_EMAILS is empty');

  // 1) 제목 구성 (접두어 + 폼키/신청자/행번호 등)
  const prefix = (CFG.EMAIL_SUBJECT_PREFIX || '').trim(); // 예: "[알림]"
  const applicant = (payload?.name || payload?.applicant || '').trim();
  const rowLabel = savedRow ? `#${savedRow}` : '';
  const pieces = [
    prefix, 
    formKey || '',
    applicant ? `- ${applicant}` : '',
    rowLabel
  ].filter(Boolean);
  const subject = pieces.join(' ').replace(/\s+/g, ' ').trim() || '알림';

  // 2) 본문 HTML
  const html = buildEmailHtml_(formKey, payload, savedRow);

  // 3) replyTo(있을 때) 설정
  const replyTo = (payload?.email || payload?.contactEmail || '').trim();

  MailApp.sendEmail({
    to: admins.join(','),
    subject,
    htmlBody: html,
    name: CFG.EMAIL_SENDER_NAME || '폼 알림',
    ...(replyTo ? { replyTo } : {})
  });

  // 4) 같은 제목을 var1로 사용해 알림톡 발송
  try {
    // ALT에서 접두어 제거를 원할 경우 옵션화
    const stripForAlt = !!CFG.STRIP_PREFIX_IN_ALT; // true면 접두어 제거
    const titleForAlt = stripForAlt && prefix
      ? subject.replace(new RegExp('^' + escapeRegExp(prefix) + '\\s*'), '').trim()
      : subject;

    // payload에 name/phone 등이 들어있어야 함
    sendAltInlineFromPayload_(payload, titleForAlt);
  } catch (e) {
    console.error('[ALT_TEST] 실패:', e);
  }
}

// 유틸: 정규식 이스케이프
function escapeRegExp(s) {
  return String(s).replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

/** ========= 이메일 HTML ========= */
function buildEmailHtml_(formKey, payload, savedRow) {
  const rows = Object.keys(payload).map(k => {
    const v = String(payload[k] || '');
    return `<tr>
      <td style="padding:6px 10px;border:1px solid #e5e7eb;background:#f9fafb;"><b>${escapeHtml_(k)}</b></td>
      <td style="padding:6px 10px;border:1px solid #e5e7eb;">${escapeHtml_(v).replace(/\n/g,'<br>')}</td>
    </tr>`;
  }).join('');
  return `<div style="font-family:system-ui,Segoe UI,Arial;">
    <h3 style="margin:0 0 8px">📨 새 신청 접수: ${escapeHtml_(formKey)}</h3>
    <div style="color:#6b7280;font-size:12px;margin-bottom:8px">저장 행: ${savedRow||'-'}</div>
    <table style="border-collapse:collapse;border:1px solid #e5e7eb;font-size:13px"><tbody>${rows}</tbody></table>
  </div>`;
}
