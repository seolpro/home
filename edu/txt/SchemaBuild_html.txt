  <script>
    /* ---------- 안전 유틸 ---------- */
    function toast(msg){ try{ alert(String(msg)); }catch(e){} }
    const $ = id => document.getElementById(id);
    function req(id){
      const el = $(id);
      if (!el) { toast(`#${id} 요소를 찾지 못했습니다. (HTML id 확인)`); throw new Error(`#${id} not found`); }
      return el;
    }
    function slugify(s){
      s = String(s||'').trim().toLowerCase()
        .normalize('NFKD').replace(/[^\w]+/g,'_').replace(/^_+|_+$/g,'');
      return s || ('field_' + (rows.length+1));
    }

    /* ---------- 상태 ---------- */
    // 기본 허용 타입(서버에서 덮어쓸 수 있음)
    let ALLOWED_TYPES = [
      'text','tel','email','number','textarea','select','address','signature',
      'date','time','datetime','month',
      'radio','checkboxes','checkbox',
      'url','password',
      'meta'
    ];
    let HEADER_FROM_ROW = 5;
    const rows = [];

    /* ---------- 렌더 ---------- */
    function redraw(){
      const tb = req('grid').querySelector('tbody');
      tb.innerHTML = '';
      rows.sort((a,b)=>Number(a.order)-Number(b.order));
      rows.forEach((r, i)=>{
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td class="text-muted row-handle">↕</td>
          <td><input type="number" class="form-control form-control-sm" value="${r.order}" data-k="order"></td>
          <td><input type="text"   class="form-control form-control-sm" value="${r.label}" data-k="label"></td>
          <td>
            <select class="form-select form-select-sm" data-k="type">
              ${ALLOWED_TYPES.map(t => `<option ${t===r.type?'selected':''}>${t}</option>`).join('')}
            </select>
          </td>
          <td><input type="text" class="form-control form-control-sm" value="${r.name}" data-k="name"></td>
          <td><input type="text" class="form-control form-control-sm" value="${r.options||''}" data-k="options"
                     ${(['select','radio','checkboxes'].includes(r.type))?'':'disabled'}></td>
          <td class="text-center"><input type="checkbox" ${r.required?'checked':''} data-k="required"></td>
          <td class="text-end">
            <button type="button" class="btn btn-sm btn-outline-danger" data-act="del">✖</button>
          </td>
        `;
        tr.querySelectorAll('input,select').forEach(el=>{
          el.onchange = ()=>{
            const k = el.dataset.k;
            let v = (k==='required') ? el.checked : el.value;
            if (k==='type') {
              const optInput = tr.querySelector('[data-k="options"]');
              optInput.disabled = !(['select','radio','checkboxes'].includes(v));
            }
            r[k] = (k==='order') ? Number(v) : v;
          };
        });
        tr.querySelectorAll('button[data-act]').forEach(btn=>{
          btn.onclick = ()=>{
            const act = btn.dataset.act;
            if (act==='del') { rows.splice(i,1); redraw(); return; }
          };
        });
        tb.appendChild(tr);
      });
    }

    /* ---------- 행 추가 ---------- */
    function addRowFromForm(){
      const orderEl   = req('f_order');
      const labelEl   = req('f_label');
      const typeEl    = req('f_type');
      const nameEl    = req('f_name');
      const optionsEl = req('f_options');
      const reqEl     = req('f_required');

      const r = {
        order: Number(orderEl.value || (rows.length ? Number(rows[rows.length-1].order)+1 : 1)),
        label: labelEl.value.trim(),
        type : typeEl.value.trim(),
        name : nameEl.value.trim() || slugify(labelEl.value),
        options: optionsEl.value.trim(),
        required: reqEl.checked
      };
      if (!r.label) { toast('label을 입력하세요.'); return; }
      if (!ALLOWED_TYPES.includes(r.type)) { toast('허용되지 않은 type 입니다.'); return; }
      if (!r.name)  { toast('name을 입력하세요.'); return; }
      if (['select','radio','checkboxes'].includes(r.type) && !r.options) {
        toast(`${r.type}는 options가 필요합니다.`); return;
      }

      rows.push(r);
      labelEl.value = ''; nameEl.value = ''; optionsEl.value = '';
      reqEl.checked = false;
      orderEl.value = String(Number(r.order)+1);
      typeEl.value = 'text'; optionsEl.disabled = true;
      redraw();
    }

    /* ---------- 로드/저장/동기화 ---------- */
    function getChosenFormKey(){
      const pick = (req('formKeyNew').value || '').trim() || (req('formKey').value || '').trim();
      if (!pick) { toast('formKey를 선택/입력하세요.'); return null; }
      return pick;
    }

    function loadForFormKey(formKey){
      if (!(window.google && google.script && google.script.run)) { redraw(); return; }
      google.script.run.withSuccessHandler(res=>{
        if (!res.ok) { toast(res.error||'로드 실패'); return; }
        if (Array.isArray(res.allowedTypes) && res.allowedTypes.length) ALLOWED_TYPES = res.allowedTypes;
        HEADER_FROM_ROW = res.headerFromRow || HEADER_FROM_ROW;

        // formKey 셀렉트 채움
        const sel = req('formKey'); sel.innerHTML = '';
        (res.formKeys||[]).forEach(k=>{
          const opt = document.createElement('option');
          opt.textContent = k; opt.value = k;
          if (k === formKey) opt.selected = true;
          sel.appendChild(opt);
        });

        rows.length = 0;
        (res.rows||[]).forEach(r=>rows.push(r));
        req('f_order').value = rows.length ? (Number(rows[rows.length-1].order)+1) : 1;
        redraw();
      }).withFailureHandler(err=>toast(err && err.message ? err.message : String(err))).sb_getSchema(formKey);
    }

    function save(){
      const formKey = getChosenFormKey(); if (!formKey) return;
      for (let i=0;i<rows.length;i++){
        const r=rows[i];
        if (!r.label) { toast(`${i+1}행 label이 비었습니다.`); return; }
        if (!r.name)  { toast(`${i+1}행 name이 비었습니다.`); return; }
        if (!ALLOWED_TYPES.includes(r.type)) { toast(`${i+1}행 type이 허용 목록에 없습니다.`); return; }
        if (['select','radio','checkboxes'].includes(r.type) && !r.options) {
          toast(`${i+1}행 options 필요(${r.type})`); return;
        }
      }
      if (!(window.google && google.script && google.script.run)) { toast('Apps Script 환경에서만 저장됩니다.'); return; }
      alert('🤷‍♂️설정하신대로 저장됩니다. 잠시만 기다려 주세요');
      google.script.run.withSuccessHandler(res=>{
        if (!res.ok) { toast(res.error||'저장 실패'); return; }
        toast(`저장 완료: ${res.count}행`);
      }).withFailureHandler(err=>toast(err && err.message ? err.message : String(err))).sb_saveSchema(formKey, rows);
    }

    function syncJoin(){
      if (!(window.google && google.script && google.script.run)) { toast('Apps Script 환경에서만 동기화됩니다.'); return; }
      alert('🤷‍♂️접수 시트의 타이틀을 동기화 합니다. 잠시만 기다려 주세요');
      google.script.run.withSuccessHandler(()=>toast('헤더 동기화/서식 완료'))
        .withFailureHandler(err=>toast(err && err.message ? err.message : String(err)))
        .syncAndStyleJoinHeaders();
    }

    /* ---------- 바인딩 ---------- */
    document.addEventListener('DOMContentLoaded', () => {
      req('btnAdd').onclick  = addRowFromForm;
      req('btnSave').onclick = save;
      req('btnSync').onclick = syncJoin;
      req('btnLoad').onclick = ()=>{ const k = getChosenFormKey(); if(k) loadForFormKey(k); };

      req('f_label').addEventListener('blur', ()=>{
        if (!req('f_name').value.trim()) req('f_name').value = slugify(req('f_label').value);
      });
      req('f_type').addEventListener('change', ()=>{
        req('f_options').disabled = !(['select','radio','checkboxes'].includes(req('f_type').value));
      });

      // 초기 로드
      if (window.google && google.script && google.script.run) {
        google.script.run.withSuccessHandler(res=>{
          const guess = (res.formKeys && res.formKeys[0]) || 'join2025';
          loadForFormKey(guess);
        }).sb_getSchema('join2025');
      } else {
        // 로컬 테스트용
        req('f_order').value = 1;
        redraw();
      }
    });
  </script>
  <script>
  // ===== 알림 설정 UI 로드/저장 =====
function setIfPresent(id, v) {
  const el = document.getElementById(id);
  if (!el) return;
  const s = (v == null ? '' : String(v).trim());
  el.value = s;             // 값이 빈 문자열이면 그대로 비워둠 → placeholder가 보임
}

function loadNotifyConfigUI() {
  if (!(window.google && google.script && google.script.run)) return;
  google.script.run.withSuccessHandler(res=>{
    if (!res || !res.ok) { toast('설정 로드 실패'); return; }
    const c = res.cfg || {};

    // 스위치
    req('cfg_enable_email').checked = !!c.ENABLE_EMAIL_NOTIFY;
    req('cfg_enable_at').checked    = !!c.ENABLE_ALIMTALK_NOTIFY;

    // 텍스트 입력들 (값 있을 때만)
    setIfPresent('cfg_admin_emails',  c.ADMIN_EMAILS);
    setIfPresent('cfg_email_prefix',  c.EMAIL_SUBJECT_PREFIX);
    setIfPresent('cfg_email_sender',  c.EMAIL_SENDER_NAME);

    // ⚠️ 기본값 넣지 마세요. placeholder만 안내하고 값은 비워둡니다.
    setIfPresent('cfg_phone_field',   c.SUBMITTER_PHONE_FIELD);

    setIfPresent('cfg_ppurio_account',   c.PPURIO_ACCOUNT);
    setIfPresent('cfg_ppurio_senderkey', c.PPURIO_SENDER_KEY);
    setIfPresent('cfg_ppurio_from',      c.PPURIO_FROM);
    setIfPresent('cfg_ppurio_tpl_code',  c.PPURIO_TEMPLATE_CODE);

    // 템플릿 본문도 placeholder로 안내하고, 저장된 값만 채움
    setIfPresent('cfg_ppurio_tpl_msg',   c.PPURIO_TEMPLATE_MESSAGE);

    req('cfg_pwd_hint').textContent =
      c.has_PPURIO_PASSWORD
        ? '저장된 비밀번호가 있습니다(미표시). 새 값을 입력하면 교체됩니다.'
        : '비밀번호가 아직 저장되어 있지 않습니다.';
    req('cfg_ppurio_password').value = ''; // 항상 비움
  }).withFailureHandler(err=>toast(err && err.message ? err.message : String(err)))
    .getNotifyConfig();
}

 function saveNotifyConfigUI() {
    if (!(window.google && google.script && google.script.run)) { toast('Apps Script 환경에서만 가능합니다.'); return; }
    const cfg = {
      ENABLE_EMAIL_NOTIFY:    req('cfg_enable_email').checked,
      ENABLE_ALIMTALK_NOTIFY: req('cfg_enable_at').checked,
      ADMIN_EMAILS:           req('cfg_admin_emails').value.trim(),
      EMAIL_SUBJECT_PREFIX:   req('cfg_email_prefix').value.trim(),
      EMAIL_SENDER_NAME:      req('cfg_email_sender').value.trim(),
      SUBMITTER_PHONE_FIELD:  req('cfg_phone_field').value.trim() || 'phone',

      PPURIO_ACCOUNT:          req('cfg_ppurio_account').value.trim(),
      PPURIO_PASSWORD:         req('cfg_ppurio_password').value, // 빈 값이면 서버에서 무시
      PPURIO_SENDER_KEY:       req('cfg_ppurio_senderkey').value.trim(),
      PPURIO_FROM:             req('cfg_ppurio_from').value.trim(),
      PPURIO_TEMPLATE_CODE:    req('cfg_ppurio_tpl_code').value.trim(),
      PPURIO_TEMPLATE_MESSAGE: req('cfg_ppurio_tpl_msg').value.trim()
    };
    alert("📤이메일(알림톡) 설정이 저장됩니다.");
    google.script.run.withSuccessHandler(res=>{
      if (!res || !res.ok) { toast(res && res.error ? res.error : '설정 저장 실패'); return; }
      toast('설정 저장 완료');
      loadNotifyConfigUI(); // 저장 후 갱신
    }).withFailureHandler(err=>toast(err && err.message ? err.message : String(err)))
      .saveNotifyConfig(cfg);
  }

  // 기존 DOMContentLoaded 바인딩에 추가
  document.addEventListener('DOMContentLoaded', () => {
    // ...
    if (window.google && google.script && google.script.run) {
      loadNotifyConfigUI();
    }
    req('btnCfgReload').onclick = loadNotifyConfigUI;
    req('btnCfgSave').onclick   = saveNotifyConfigUI;
  });
</script>
